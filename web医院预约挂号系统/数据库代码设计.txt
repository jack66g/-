-- 创建医院数据库
CREATE DATABASE IF NOT EXISTS `hospital_db` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- 使用数据库
USE `hospital_db`;

-- 创建患者信息表（所有敏感字段加密存储）
CREATE TABLE IF NOT EXISTS `patient_info` (
  `patient_id` INT AUTO_INCREMENT PRIMARY KEY COMMENT '患者ID',
  `name` VARCHAR(30) NOT NULL COMMENT '姓名（1-10汉字）',
  `gender` VARCHAR(3) NOT NULL COMMENT '性别（1汉字）',
  `age` VARBINARY(128) NOT NULL COMMENT '年龄（AES加密）',
  `phone` VARBINARY(128) NOT NULL COMMENT '电话（AES加密）',
  `password` VARBINARY(128) NOT NULL COMMENT '密码（PBKDF2加密）',
  `id_card` VARBINARY(128) NOT NULL COMMENT '身份证（AES加密）',
  `address` VARBINARY(256) NOT NULL COMMENT '地址（AES加密）',
  `medical_card` VARBINARY(128) NOT NULL COMMENT '医保卡号（AES加密）',
  -- 加密相关元数据
  `encryption_iv` VARBINARY(16) NOT NULL COMMENT 'AES加密IV',
  `salt` VARBINARY(64) NOT NULL COMMENT '密码盐值'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 添加字段注释（MySQL 8.0+ 支持）
ALTER TABLE `patient_info` 
  MODIFY COLUMN `name` VARCHAR(30) COMMENT '姓名（1-10汉字）',
  MODIFY COLUMN `gender` VARCHAR(3) COMMENT '性别（1汉字）';
ALTER TABLE `patient_info`
    ADD COLUMN `is_registered` TINYINT(1) NOT NULL DEFAULT 0 
    COMMENT '挂号状态（1=已挂号，0=未挂号）'
    AFTER `salt`;

管理员数据库，仅限后台插入新的管理员，其他方式不提供
-- 创建管理员表
CREATE TABLE IF NOT EXISTS `hospital_db`.`admin_account` (
  `admin_id` INT(6) UNSIGNED ZEROFILL PRIMARY KEY COMMENT '管理员ID（6位数字）',
  `account` VARCHAR(11) NOT NULL COMMENT '账号（11位数字）',
  `password` VARCHAR(20) NOT NULL COMMENT '密码（明文存储）',
  
  -- 添加数据校验规则
  CONSTRAINT chk_account_length CHECK (CHAR_LENGTH(account) = 11),
  CONSTRAINT chk_password_length CHECK (CHAR_LENGTH(password) BETWEEN 8 AND 20)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

医生信息表创建
-- 创建医生信息表（含图片二进制存储）
CREATE TABLE IF NOT EXISTS `hospital_db`.`doctor_info` (
  `doctor_id` CHAR(6) PRIMARY KEY COMMENT '医生工号（6位纯数字）',
  `doctor_name` VARCHAR(30) NOT NULL COMMENT '医生姓名（30字符）',
  `photo_data` MEDIUMBLOB NOT NULL COMMENT '压缩图片数据（JPEG格式）',
  `introduction` TEXT COMMENT '医生简介（可空）',
  `source_1` CHAR(2) NOT NULL COMMENT '号源1（两位数字）',
  `source_2` CHAR(2) NOT NULL COMMENT '号源2（两位数字）',
  `source_3` CHAR(2) NOT NULL COMMENT '号源3（两位数字）',
  `source_4` CHAR(2) NOT NULL COMMENT '号源4（两位数字）',

  -- 约束条件
  CONSTRAINT chk_doctor_id CHECK (doctor_id REGEXP '^[0-9]{6}$'),
  CONSTRAINT chk_source_1 CHECK (source_1 REGEXP '^[0-9]{2}$'),
  CONSTRAINT chk_source_2 CHECK (source_2 REGEXP '^[0-9]{2}$'),
  CONSTRAINT chk_source_3 CHECK (source_3 REGEXP '^[0-9]{2}$'),
  CONSTRAINT chk_source_4 CHECK (source_4 REGEXP '^[0-9]{2}$')
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
